namespace Laba5;

public class MultiList
{
    private readonly StudentHash _students = new StudentHash();
    private readonly CourseHash _courses = new CourseHash();

    public void AddStudent(char[] name) => _students.Insert(name);
    public void AddCourse(char[] name) => _courses.Insert(name);
    
    // 1. Добавить студента на курс
    public void AddStudentToCourse(char[] sName, char[] cName)
    {
        StudentNode? student = _students.GetNode(sName);
        CourseNode? course = _courses.GetNode(cName);
        if (student == null || course == null) return;

        // Проверка на существование связи
        if (FindRegistration(student, course) != null) return;

        RegistrationNode reg = new RegistrationNode();

        // Вставка в кольцо студента (в начало)
        if (student.Next == null)
        {
            student.Next = reg;
            reg.StudentLink = student;
        }
        else
        {
            reg.StudentLink = student.Next;
            student.Next = reg;
        }

        // Вставка в кольцо курса (в начало)
        if (course.Next == null)
        {
            course.Next = reg;
            reg.CourseLink = course;
        }
        else
        {
            reg.CourseLink = course.Next;
            course.Next = reg;
        }
    }

    // 2. Удалить студента с конкретного курса
    public void RemoveStudentFromCourse(char[] sName, char[] cName)
    {
        StudentNode? student = _students.GetNode(sName);
        CourseNode? course = _courses.GetNode(cName);
        if (student == null || course == null) return;

        RegistrationNode? reg = FindRegistration(student, course);
        if (reg == null) return;

        RemoveFromStudentRing(reg, student);
        RemoveFromCourseRing(reg, course);
    }

    // 3. Удалить студента со всех курсов
    public void RemoveStudentFromAll(char[] sName)
    {
        StudentNode? student = _students.GetNode(sName);
        if (student?.Next == null) return;

        BaseNode? cur = student.Next;
        while (cur != student && cur != null)
        {
            RegistrationNode reg = (RegistrationNode)cur;
            BaseNode? next = reg.StudentLink;

            // Находим курс этой регистрации (идём по CourseLink до CourseNode)
            CourseNode? course = GetCourseFromRegistration(reg);
            
            // CourseNode? course = null;
            // BaseNode? c = reg.CourseLink;
            // while (c != null && c.IsRegistration())
            // {
            //     c = ((RegistrationNode)c).CourseLink;
            // }
            // course = c as CourseNode;

            if (course != null)
                RemoveFromCourseRing(reg, course);
            cur = next;
        }
        student.Next = null;
    }

    // 4. Удалить всех студентов с конкретного курса
    public void RemoveAllFromCourse(char[] cName)
    {
        CourseNode? course = _courses.GetNode(cName);
        if (course?.Next == null) return;

        BaseNode? cur = course.Next;
        while (cur != course && cur != null)
        {
            RegistrationNode reg = (RegistrationNode)cur;
            BaseNode? next = reg.CourseLink;

            // Находим студента этой регистрации (идём по StudentLink до StudentNode)
            
            StudentNode? student = GetStudentFromRegistration(reg);
            
            // StudentNode? student = null;
            // BaseNode? s = reg.StudentLink;
            // while (s != null && s.IsRegistration())
            // {
            //     s = ((RegistrationNode)s).StudentLink;
            // }
            // student = s as StudentNode;

            if (student != null)
                RemoveFromStudentRing(reg, student);

            cur = next;
        }
        course.Next = null;
    }

    // 5. Вывести курсы, которые посещает студент
    public void PrintCoursesOfStudent(char[] sName)
    {
        StudentNode? student = _students.GetNode(sName);
        if (student?.Next == null)
        {
            Console.WriteLine("Студент не найден или не записан на курсы.");
            return;
        }

        Console.Write("Курсы студента ");
        PrintName(sName);
        Console.WriteLine(":");

        BaseNode? cur = student.Next;
        while (cur != student && cur != null)
        {
            RegistrationNode reg = (RegistrationNode)cur;

            // Ищем курс регистрации (локальный цикл)
            CourseNode? course = GetCourseFromRegistration(reg);

            if (course != null)
            {
                PrintName(course.Name);
                Console.WriteLine();
            }
            cur = reg.StudentLink;
        }
    }

    // 6. Вывести студентов, посещающих курс
    public void PrintStudentsOfCourse(char[] cName)
    {
        CourseNode? course = _courses.GetNode(cName);
        if (course?.Next == null)
        {
            Console.WriteLine("Курс не найден или нет записанных студентов.");
            return;
        }

        Console.Write("Студенты курса ");
        PrintName(cName);
        Console.WriteLine(":");

        BaseNode? cur = course.Next;
        while (cur != course && cur != null)
        {
            RegistrationNode reg = (RegistrationNode)cur;

            // Ищем студента регистрации (локальный цикл)
            StudentNode? student = GetStudentFromRegistration(reg);
            
            // BaseNode? s = reg.StudentLink;
            // while (s != null && s.IsRegistration())
            //     s = ((RegistrationNode)s).StudentLink;
            // StudentNode? student = s as StudentNode;

            if (student != null)
            {
                PrintName(student.Name);
                Console.WriteLine();
            }
            cur = reg.CourseLink;
        }
    }

    // ==================== Вспомогательные методы ====================

    // Поиск регистрации, связывающей заданных студента и курс.
    private RegistrationNode? FindRegistration(StudentNode student, CourseNode course)
    {
        BaseNode? cur = student.Next;
        while (cur != null && cur != student)
        {
            RegistrationNode reg = (RegistrationNode)cur;

            // Определяем курс этой регистрации (локальный обход до CourseNode)
            BaseNode? c = reg.CourseLink;
            while (c != null && c.IsRegistration())
                c = ((RegistrationNode)c).CourseLink;
            if (c == course)
                return reg;

            cur = reg.StudentLink;
        }
        return null;
    }

    // Удаление регистрации из кольца студента (студент передаётся параметром)
    private void RemoveFromStudentRing(RegistrationNode reg, StudentNode student)
    {
        if (student.Next == reg)
        {
            student.Next = (reg.StudentLink != student) ? reg.StudentLink : null;
            return;
        }

        BaseNode? prev = FindPrevInStudentRing(reg, student);
        if (prev is RegistrationNode prevReg)
            prevReg.StudentLink = reg.StudentLink;
    }

    // Удаление регистрации из кольца курса (курс передаётся параметром)
    private void RemoveFromCourseRing(RegistrationNode reg, CourseNode course)
    {
        if (course.Next == reg)
        {
            course.Next = (reg.CourseLink != course) ? reg.CourseLink : null;
            return;
        }

        BaseNode? prev = FindPrevInCourseRing(reg, course);
        if (prev is RegistrationNode prevReg)
            prevReg.CourseLink = reg.CourseLink;
    }

    private CourseNode? GetCourseFromRegistration(RegistrationNode reg)
    {
        BaseNode? cur = reg.CourseLink;
        while (cur != null && cur.IsRegistration())
            cur = ((RegistrationNode)cur).CourseLink;
        return cur as CourseNode;
    }

    private StudentNode? GetStudentFromRegistration(RegistrationNode reg)
    {
        BaseNode? cur = reg.StudentLink;
        while (cur != null && cur.IsRegistration())
            cur = ((RegistrationNode)cur).StudentLink;
        return cur as StudentNode;
    }
    
    // Поиск предыдущего узла в кольце студента перед target
    private BaseNode? FindPrevInStudentRing(RegistrationNode target, StudentNode student)
    {
        BaseNode? cur = student.Next;
        BaseNode? prev = student;
        while (cur != student && cur != null)
        {
            if (cur == target)
                return prev;
            prev = cur;
            if (cur.IsRegistration())
                cur = ((RegistrationNode)cur).StudentLink;
            else break;
        }
        return null;
    }

    // Поиск предыдущего узла в кольце курса перед target
    private BaseNode? FindPrevInCourseRing(RegistrationNode target, CourseNode course)
    {
        BaseNode? cur = course.Next;
        BaseNode? prev = course;
        while (cur != course && cur != null)
        {
            if (cur == target)
                return prev;
            prev = cur;
            if (cur.IsRegistration())
                cur = ((RegistrationNode)cur).CourseLink;
            else break;
        }
        return null;
    }

    // Вывод имени (char[]) в консоль
    private static void PrintName(char[] name)
    {
        int i = 0;
        while (i < name.Length && name[i] != '\0')
        {
            Console.Write(name[i]);
            i++;
        }
    }
}
